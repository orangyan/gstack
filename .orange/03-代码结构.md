```typescript
      if (url.pathname === '/command') return handleCommand(req);

      if (url.pathname === '/sidebar-chat') return handleSidebarChat(req);
```

```typescript
// 可访问性树 + @ref 令牌系统
export async function snapshot(page: Page, flags: string[]): Promise<string> {
  // 1. 获取可访问性树
  const tree = await page.accessibility.snapshot();

  // 2. 转换为带 @ref 令牌的文本
  const output = formatTree(tree, {
    includeInteractive: flags.includes('-i'),
    includeText: flags.includes('-t'),
    includeIds: flags.includes('--ids'),
    includeRoles: flags.includes('--roles'),
  });

  // 3. 存储 @ref 映射用于后续命令
  storeRefMap(page, tree);

  return output;
}

// @ref 令牌格式：@e3 = 元素 3，@c1 = 容器 1
function formatTree(node: AccessibilityNode, opts: SnapshotOptions): string {
  let result = '';
  if (node.role) result += `${node.role}`;
  if (node.name) result += `: ${node.name}`;
  if (opts.includeInteractive && node.focusable) result += ` @e${nextElementId++}`;
  // ... 递归格式化子节点
  return result;
}
```

```typescript
// L1-L3：数据标记 + 隐藏元素 + ARIA + URL 黑名单
export function markHiddenElements(page: Page): Promise<string[]> {
  return page.evaluate(() => {
    const hidden: string[] = [];
    document.querySelectorAll('*').forEach(el => {
      const style = window.getComputedStyle(el);
      if (style.display === 'none' || style.visibility === 'hidden') {
        el.setAttribute('data-hidden-element', 'true');
        hidden.push(el.tagName + (el.id ? `#${el.id}` : ''));
      }
    });
    return hidden;
  });
}

export function stripHiddenElements(html: string): string {
  return html.replace(/<[^>]*data-hidden-element="true"[^>]*>.*?<\/[^>]*>/gs, '');
}

export const DANGEROUS_ARIA_PATTERNS = [
```

```typescript
import OpenAI from 'openai';

export async function generate(prompt: string, opts: GenerateOptions) {
  const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

  // 1. 调用 GPT Image API
  const response = await openai.images.generate({
    model: 'dall-e-3',
    prompt: prompt,
    size: opts.size || '1024x1024',
    quality: opts.quality || 'standard',
    n: 1,
  });

  // 2. 下载图像
  const imageUrl = response.data[0].url;
  const imageData = await fetch(imageUrl).then(r => r.arrayBuffer());

  // 3. 保存到磁盘
  const filename = `design-${Date.now()}.png`;
  await Bun.write(filename, imageData);

  console.log(`Saved: ${filename}`);
}
```

```typescript
export async function variants(prompt: string, count: number) {
  // 1. 加载品味档案（如果存在）
  const taste = await loadTasteProfile();

  // 2. 增强提示与品味偏好
  const enhancedPrompt = applyTasteProfile(prompt, taste);

  // 3. 生成多个变体（并行）
  const promises = Array.from({ length: count }, (_, i) =>
    generate(`${enhancedPrompt} (variant ${i + 1})`, opts)
  );

  const results = await Promise.all(promises);

  // 4. 启动比较服务器
  await serve(results.map(r => r.filename));
}
```

**品味学习**：`bin/gstack-taste-update`

```typescript
#!/usr/bin/env bun
// 从 /design-shotgun 批准/拒绝写入持久化品味档案
import { readFileSync, writeFileSync } from 'fs';

const TASTE_FILE = '~/.gstack/taste-profile.json';
const DECAY_RATE = 0.05;  // 每周 5%

function updateTaste(approval: TasteApproval) {
  const profile = loadTasteProfile();

  // 更新偏好
  for (const [key, value] of Object.entries(approval)) {
    if (profile[key]) {
      profile[key].weight += value > 0 ? 0.1 : -0.1;
    } else {
      profile[key] = { weight: value > 0 ? 0.1 : -0.1, lastUpdated: Date.now() };
    }
  }

  // 应用衰减到旧偏好
  const now = Date.now();
  for (const [key, pref] of Object.entries(profile)) {
    const weeksSince = (now - pref.lastUpdated) / (7 * 24 * 60 * 60 * 1000);
    pref.weight *= Math.pow(1 - DECAY_RATE, weeksSince);
  }

  writeFileSync(TASTE_FILE, JSON.stringify(profile, null, 2));
}
```

export interface ParsedSkill {
  name: string;
  description: string;
  commands: Command[];
  examples: Example[];
  usage: string;
  flags: string[];
}

export function parseSkill(path: string): ParsedSkill {
  const content = readFileSync(path, 'utf-8');

  // 1. 提取名称（来自第一个 # 标题）
  const nameMatch = content.match(/^# (.+)$/m);
  const name = nameMatch ? nameMatch[1] : '';

  // 2. 提取描述（来自第一段）
  const descMatch = content.match(/^# .+\n\n(.+?)\n\n/s);
  const description = descMatch ? descMatch[1] : '';

  // 3. 提取命令（来自 ```bash 代码块）
  const commands: Command[] = [];
  const commandRegex = /```bash\n(.+?)\n```/gs;
  let match;
  while ((match = commandRegex.exec(content)) !== null) {
    commands.push(parseCommand(match[1]));
  }

  // 4. 提取示例（来自 ## Examples 节）
  const examples: Example[] = [];
  const examplesSection = content.match(/## Examples\n\n(.+?)(?=\n##|$)/s);
  if (examplesSection) {
    // ... 解析示例
  }

  return { name, description, commands, examples, usage: '', flags: [] };
}

**解析器示例**：

```typescript
// scripts/resolvers/my-resolver.ts
export function resolveMyResolver(skill: Skill, host: string): string {
  // 1. 加载数据
  const data = loadData();

  // 2. 过滤为相关
  const relevant = data.filter(d => d.skill === skill.name);

  // 3. 格式化为 Markdown
  return `
## My Custom Section

${relevant.map(r => `- ${r.name}: ${r.description}`).join('\n')}
  `;
}

// scripts/gen-skill-docs.ts
const RESOLVERS: Record<string, ResolverFunction> = {
  // ...
  '{{MY_RESOLVER}}': resolveMyResolver,
};
```

## 六、总结

GStack 的代码结构体现了三个核心原则：

1. **单一真相来源** - `commands.ts` 用于命令，`hosts/index.ts` 用于主机，`.tmpl` 文件用于技能
2. **配置驱动** - 添加新主机/技能/命令无需代码更改，仅配置
3. **测试优先** - 每个技能都有 E2E 测试，每个命令都有单元测试

这些原则使单个开发者能够维护一个支持 10+ AI 代理、23+ 专家技能和 100+ 浏览器命令的项目 - 不是通过更努力工作，而是通过正确的抽象和自动化。
